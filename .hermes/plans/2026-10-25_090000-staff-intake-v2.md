Read `2026-10-25_090000-hc-staff-identity-roadmap.md` first — it carries the source register, the verified defect list D1–D14, and decisions X1–X10 that this plan implements. **Goal:** Make the five-step staff intake actually work end to end — navigable steps, a finalization boundary that cannot create a half-employee, real private attachments, a complete Digital Staff Profile, a scoped Excel import UI, and account activation without emailed passwords.

**Architecture:** Keep `sdm.Pegawai` as the canonical identity. Move finalization out of `sdm/intake_views.py` into `sdm/services/intake.py` as the single boundary; the view becomes a thin adapter. Stage uploaded files as `LampiranDrafPegawai` rows against the draft, then bind them to typed `DokumenPegawai`/`VersiDokumen` rows at finalization so a partial upload can never claim a saved employee. Photo handling normalizes and bounds the image before storage. Account invitation stays optional, post-commit, and activation-link based.

## Review focus

These are the failure modes a reasonable operator would hit that the source document does not mention. Each has a test in the task that owns the code.

1. A draft left open in two tabs, saved from the stale one, silently discarding the newer step data.
2. A draft finalized while its operator's grant was revoked mid-session, producing an employee in a unit the operator no longer controls.
3. A photo upload of a 12-megapixel phone image stored and later served as a directory avatar, stalling the page.
4. A failed storage upload leaving an employee with a document row that has no file, which then counts as "uploaded" in compliance percentages.
5. An activation link that works twice, or that attaches an unrelated pre-existing account that happens to share the email.
6. A `nik` field submitted as `'0000000000000001'` losing its leading zeros through an integer conversion.

from decimal import Decimal

from django.core.exceptions import PermissionDenied, ValidationError
from django.test import TestCase

STEP1 = {
    'nama_depan': 'Budi', 'nama_belakang': 'Santoso',
    'nik': '3201010101010001', 'telepon': '081234567890',
    'email_terverifikasi': 'budi@example.com',

class IntakeFinalizationTests(TestCase):
    def setUp(self):
        self.today = timezone.localdate()
        self.root = UnitKerja.objects.create(name='Direktorat HC', code='INT-HC')
        self.other = UnitKerja.objects.create(name='Unit Rahasia', code='INT-OTHER')
        self.manager = self._user('intake-manager')

    def _user(self, username):
        user = get_user_model().objects.create_user(username=username)
        UserProfile.objects.create(user=user, role='ADMIN_RS', unit_kerja=self.root)
        config = RolePermissionConfig.get_config_for_role('ADMIN_RS')
        config.can_view_hc = True
        config.can_manage_hc_employees = True
        config.save(update_fields=['can_view_hc', 'can_manage_hc_employees', 'updated_at'])
        HcGrant.objects.create(user=user, capability='can_view_hc', unit_kerja=self.root)
        HcGrant.objects.create(
            user=user, capability='can_manage_hc_employees', unit_kerja=self.root
        )
        return user

    def _draft(self, *, steps=('1', '2', '3', '4', '5'), version=1, unit=None,
               fte='1.00', **overrides):
        payload = {
            '1': dict(STEP1),
            '2': {
                'nomor_pegawai': 'INT-001', 'kategori': 'NON_PPA',
                'profesi': 'Administrasi', 'jabatan': 'Staf', 'status': 'AKTIF',
                'kategori_pegawai': 'TETAP', 'unit_kerja': (unit or self.root).pk,
                'tanggal_mulai': self.today.isoformat(), 'alokasi_fte': fte,
            },
            '3': {},
            '4': {},
            '5': {},
        }
        payload.update(overrides)
        return DrafInputPegawai.objects.create(
            owner=self.manager,
            current_step=5,
            version=version,
            expires_at=timezone.now() + timedelta(days=1),
            draft_data={key: payload[key] for key in steps},
        )

    def test_all_five_steps_are_required(self):
        draft = self._draft(steps=('1', '2'))
        with self.assertRaises(ValidationError):
            finalize_staff_intake(
                actor=self.manager, draft_id=draft.pk,
                expected_version=1, idempotency_key='k-1',
            )
        self.assertFalse(Pegawai.objects.exists())

    def test_blank_but_present_optional_step_is_complete(self):
        draft = self._draft()  # steps 3/4/5 present and empty
        employee = finalize_staff_intake(
            actor=self.manager, draft_id=draft.pk,
            expected_version=1, idempotency_key='k-2',
        )
        self.assertEqual(employee.nomor_pegawai, 'INT-001')

    def test_stale_version_is_refused_without_writing(self):
        draft = self._draft(version=3)
        with self.assertRaises(ValidationError):
            finalize_staff_intake(
                actor=self.manager, draft_id=draft.pk,
                expected_version=1, idempotency_key='k-3',
            )
        self.assertFalse(Pegawai.objects.exists())
        draft.refresh_from_db()
        self.assertEqual(draft.state, 'DRAFT')

    def test_foreign_unit_is_refused(self):
        draft = self._draft(unit=self.other)
        with self.assertRaises(PermissionDenied):
            finalize_staff_intake(
                actor=self.manager, draft_id=draft.pk,
                expected_version=1, idempotency_key='k-4',
            )
        self.assertFalse(Pegawai.objects.exists())

    def test_duplicate_nip_is_refused(self):
        Pegawai.objects.create(nomor_pegawai='INT-001', nama_lengkap='Existing',
                               kategori='NON_PPA')
        draft = self._draft()
        with self.assertRaises(ValidationError):
            finalize_staff_intake(
                actor=self.manager, draft_id=draft.pk,
                expected_version=1, idempotency_key='k-5',

        with self.assertRaises(ValidationError):
            finalize_staff_intake(
                actor=self.manager, draft_id=draft.pk,
                expected_version=1, idempotency_key='k-6',
            )
        self.assertFalse(Pegawai.objects.exists())

    def test_placement_goes_through_the_assignment_service(self):
        draft = self._draft(fte='0.75')
        employee = finalize_staff_intake(
            actor=self.manager, draft_id=draft.pk,
            expected_version=1, idempotency_key='k-7',
        )
        assignment = employee.penempatan.get()
        self.assertEqual(assignment.alokasi_fte, Decimal('0.75'))
        self.assertEqual(assignment.unit_kerja, self.root)

    def test_matching_replay_returns_the_same_employee(self):
        draft = self._draft()
        first = finalize_staff_intake(
            actor=self.manager, draft_id=draft.pk,
            expected_version=1, idempotency_key='k-8',
        )
        second = finalize_staff_intake(
            actor=self.manager, draft_id=draft.pk,
            expected_version=1, idempotency_key='k-8',
        )
        self.assertEqual(first.pk, second.pk)
        self.assertEqual(Pegawai.objects.count(), 1)
        self.assertEqual(PenempatanPegawai.objects.count(), 1)
        self.assertEqual(

```python
    def fetch_object(self, storage_key: str, *, max_bytes: int | None = None) -> bytes:
        """Download a private object's bytes for same-origin serving."""
        match = re.fullmatch(
            r"(personnel|health)/([0-9a-f-]{36})/([0-9a-f-]{36})\.(pdf|png|jpg|jpeg)",
            storage_key or "",
        )
        if not match:
            raise PrivateStorageError("Invalid private-object fetch request")
        limit = max_bytes if max_bytes is not None else self.max_bytes
        if type(limit) is not int or limit <= 0:
            raise PrivateStorageError("Invalid private-object fetch request")
        endpoint = (
            f"{self.base_url}/storage/v1/object/{quote(self.bucket, safe='')}/"
            f"{quote(storage_key, safe='/')}"
        )
        response = self._request(
            "GET",
            endpoint,
            headers={
                "apikey": self.api_key,
                "Authorization": f"Bearer {self.api_key}",
            },
            stream=True,
        )
        try:
            chunks = []
            total = 0
            for chunk in response.iter_content(chunk_size=65536):
                if not chunk:
                    continue
                if not isinstance(chunk, bytes):
                    raise PrivateStorageError("Private-object fetch failed")
                total += len(chunk)
                if total > limit:
                    raise PrivateStorageError("Private-object fetch failed")
                chunks.append(chunk)
            return b"".join(chunks)
        except PrivateStorageError:
            raise
        except Exception:
            raise PrivateStorageError("Private-object fetch failed") from None
        finally:
            close = getattr(response, "close", None)
            if close:
                close()

PHOTO_MIN_DIMENSION = 200
PHOTO_SOURCE_MAX_BYTES = 12_000_000
PHOTO_SOURCE_MAX_PIXELS = 20_000_000


def normalize_profile_photo(file_obj):
    """Return a bounded, EXIF-corrected JPEG for avatar use."""
    from io import BytesIO

    from PIL import Image, ImageOps, UnidentifiedImageError

    raw = file_obj.read(PHOTO_SOURCE_MAX_BYTES + 1)
    if not isinstance(raw, bytes) or not raw or len(raw) > PHOTO_SOURCE_MAX_BYTES:
        raise ValidationError({'foto': 'Berkas foto tidak valid atau terlalu besar.'})
    try:
        with Image.open(BytesIO(raw)) as source:
            if source.width * source.height > PHOTO_SOURCE_MAX_PIXELS:
                raise ValidationError({'foto': 'Resolusi foto terlalu besar.'})
            source.verify()
        with Image.open(BytesIO(raw)) as source:
            image = ImageOps.exif_transpose(source).convert('RGB')
    except (UnidentifiedImageError, OSError, ValueError):
        raise ValidationError({'foto': 'Berkas bukan gambar yang dapat dibaca.'}) from None
    if min(image.size) < PHOTO_MIN_DIMENSION:
        raise ValidationError({
            'foto': f'Ukuran foto minimal {PHOTO_MIN_DIMENSION}×{PHOTO_MIN_DIMENSION} piksel.'
        })
    image.thumbnail((PHOTO_MAX_DIMENSION, PHOTO_MAX_DIMENSION), Image.LANCZOS)
    buffer = BytesIO()
    image.save(buffer, format='JPEG', quality=85, optimize=True)
    return buffer.getvalue(), 'image/jpeg'

def stage_intake_attachment(*, actor, draft_id, expected_version, file_obj, artifact_kind):

    from io import BytesIO

    if type(expected_version) is not int:
        raise ValidationError({'version': 'Versi draf tidak valid.'})

        .first()

    require_hc_capability(actor, 'can_manage_hc_employees', unit=target_unit)
    require_hc_capability(actor, 'can_manage_hc_documents', unit=target_unit)
    if draft.state != 'DRAFT':
        raise ValidationError({'draft': 'Draf tidak dapat menerima berkas.'})

        raise ValidationError({'version': 'Draf telah diperbarui di tab lain.'})

        and draft.attachments.filter(
            artifact_kind=artifact_kind, state='STAGED'
        ).exists()

            'berkas': 'Lampiran jenis ini sudah ada. Hapus lampiran lama sebelum mengganti.'

    extension = raw_name.rsplit('.', 1)[-1].lower() if '.' in raw_name else ''

def bind_intake_attachments(*, draft, employee, actor):
    """Bind all staged objects inside the caller's finalization transaction."""
    from sdm.models import DokumenPegawai, PeristiwaSDM, VersiDokumen

    versions = []
    attachments = draft.attachments.select_for_update().filter(state='STAGED')
    step5 = (draft.draft_data or {}).get('5') or {}
    for attachment in attachments:
        document_type, kind, default_title = INTAKE_DOCUMENT_META[attachment.artifact_kind]
        title = default_title
        if attachment.artifact_kind not in ('FOTO_PROFIL', 'KTP'):
            title = (step5.get('dokumen_judul') or '').strip() or default_title
        document = DokumenPegawai(
            employee=employee,
            document_type=document_type,
            jenis_berkas=kind,
            title=title,
            sensitivity='personnel',
            created_by=actor,
        )
        document.full_clean()
        document.save()
        version = VersiDokumen(
            document=document,
            version_number=1,
            original_filename=attachment.original_filename,
            storage_key=attachment.storage_key,
            sha256=attachment.sha256,
            size_bytes=attachment.size_bytes,
            content_type=attachment.content_type,
            status='PENDING',
            created_by=actor,
        )
        version.full_clean()
        version.save()
        PeristiwaSDM.objects.create(
            actor=actor,
            jenis_peristiwa='private_document_uploaded',
            target_tipe='document_version',
            target_id=str(version.pk),
            status_sesudah='PENDING',
            alasan='sensitivity=personnel; source=intake',
        )
        attachment.state = 'BOUND'
        attachment.bound_version = version
        attachment.save(update_fields=['state', 'bound_version'])
        if attachment.artifact_kind == 'FOTO_PROFIL':
            profile = employee.profil_pribadi
            profile.foto_profil_versi = version
            profile.save(update_fields=['foto_profil_versi', 'updated_at'])
        versions.append(version)
    return versions

def build_staff_profile(employee, *, viewer):
    """Assemble the Digital Staff Profile with field-level permissions."""
    from django.core.exceptions import PermissionDenied

    from sdm.akses import require_hc_capability
    from sdm.services.kepatuhan import documents_for_employee

    can_view_personal = employee.akun_id == viewer.pk
    if not can_view_personal:
        try:
            require_hc_capability(
                viewer, 'can_manage_hc_employees', employee=employee
            )
            can_view_personal = True
        except PermissionDenied:
            can_view_personal = False

    personal_record = getattr(employee, 'profil_pribadi', None)
    # Never pass the restricted model instance to a template that may render it
    # for a view-only sibling. The presence flag supports an honest empty state.
    personal = personal_record if can_view_personal else None
    employment = employee.riwayat_kepegawaian.order_by('-tanggal_tmt', '-pk').first()
    education = list(
        employee.riwayat_pendidikan.select_related(
            'ijazah_versi', 'transkrip_versi'
        ).order_by('-tanggal_lulus', '-pk')
    )
    permits = list(
        employee.izin_profesi.order_by('subtype', 'nomor_izin')
    )

    nik = getattr(personal_record, 'nik', '') or ''
    masked_nik = 'Tersembunyi'
    if nik and can_view_personal:
        masked_nik = nik

    tenure_days = None
    tmt = getattr(employment, 'tanggal_tmt', None)
    if tmt:
        tenure_days = (timezone.localdate() - tmt).days

    invitation = employee.invitations.order_by('-created_at').first()

    return {
        'personal': personal,
        'employment': employment,
        'education': education,
        'permits': permits,
        'documents': documents_for_employee(employee).prefetch_related('versions'),
        'invitation': invitation if can_view_personal else None,
        'can_view_personal': can_view_personal,
        'has_personal_record': personal_record is not None,
        'masked_nik': masked_nik,
        'tenure_days': tenure_days,
    }
```

    document = version.document
    if (
        document.employee_id != employee.pk
        or document.document_type != 'IDENTITY'
        or document.jenis_berkas != 'FOTO'
        or document.sensitivity != 'personnel'
        or version.content_type != 'image/jpeg'
        or version.status not in ('PENDING', 'VERIFIED')
        or not 1 <= version.size_bytes <= 2_000_000
    ):
        raise Http404('Foto profil belum tersedia.')
    storage = get_private_storage()
    try:
        data = storage.fetch_object(version.storage_key, max_bytes=2_000_000)
    except PrivateStorageError:
        raise Http404('Foto profil tidak dapat dibaca.') from None
    if (
        len(data) != version.size_bytes
        or not data.startswith(b'\xff\xd8\xff')
        or hashlib.sha256(data).hexdigest() != version.sha256
    ):
        raise Http404('Foto profil tidak dapat dibaca.')
    response = HttpResponse(data, content_type='image/jpeg')
    response['Cache-Control'] = 'private, max-age=300'
    response['Content-Disposition'] = 'inline'
    return response

```python
def accept_staff_invitation(*, token, password):
    digest = hashlib.sha256((token or '').encode()).hexdigest()
    error = None
    accepted_user = None
    User = get_user_model()
    from accounts.models import UserProfile

    with transaction.atomic():
        invitation = (
            UndanganAkunPegawai.objects.select_for_update()
            .select_related('employee')
            .filter(token_digest=digest)
            .first()
        )
        if invitation is None or invitation.state != 'SENT':
            error = 'Tautan aktivasi tidak valid atau sudah digunakan.'
        elif timezone.now() >= invitation.expiry:
            invitation.state = 'EXPIRED'
            invitation.save(update_fields=['state', 'updated_at'])
            error = 'Tautan aktivasi tidak valid atau sudah digunakan.'
        else:
            employee = Pegawai.objects.select_for_update().get(
                pk=invitation.employee_id
            )
            canonical_email = User.objects.normalize_email(invitation.email)
            candidate = User(username=canonical_email, email=canonical_email)
            validate_password(password, user=candidate)

            if employee.akun_id is not None:
                error = 'Tautan aktivasi tidak valid atau sudah digunakan.'
            elif User.objects.filter(
                Q(username__iexact=canonical_email)
                | Q(email__iexact=canonical_email)
            ).exists():
                error = 'Tautan aktivasi tidak valid atau sudah digunakan.'
            else:
                user = User.objects.create_user(
                    username=canonical_email,
                    email=canonical_email,
                    password=password,
                    is_active=invitation.intended_is_active,
                )
                UserProfile.objects.create(
                    user=user,
                    role=invitation.intended_role,
                    unit_kerja=employee.unit_kerja,
                )
                employee.akun = user
                employee.save(update_fields=['akun', 'updated_at'])
                invitation.state = 'ACCEPTED'
                invitation.accepted_at = timezone.now()
                invitation.attempt_count = F('attempt_count') + 1
                invitation.save(update_fields=[
                    'state', 'accepted_at', 'attempt_count', 'updated_at',
                ])
                PeristiwaSDM.objects.create(
                    actor=user,
                    jenis_peristiwa='account_invitation_accepted',
                    target_tipe='employee',
                    target_id=str(employee.pk),
                    alasan=(
                        f'role={invitation.intended_role}; '
                        f'active={invitation.intended_is_active}'
                    ),
                )
                accepted_user = user

    if error:
        raise ValidationError(error)
    return accepted_user
```

```python
"""Single-use staff account activation (Task 9)."""

import hashlib

from django.core.exceptions import ValidationError
from django.shortcuts import redirect, render
from django.views.decorators.csrf import csrf_protect
from django.views.decorators.http import require_GET, require_http_methods

from sdm.models import UndanganAkunPegawai
from sdm.services.undangan import accept_staff_invitation

GENERIC_ERROR = 'Tautan aktivasi tidak valid atau sudah digunakan.'


def _render_private(request, template, context):
    response = render(request, template, context)
    response['Cache-Control'] = 'no-store'
    response['Referrer-Policy'] = 'no-referrer'
    return response


def _pending_invitation(token):
    digest = hashlib.sha256((token or '').encode()).hexdigest()
    return UndanganAkunPegawai.objects.filter(
        token_digest=digest, state='SENT'
    ).select_related('employee').first()


@csrf_protect
@require_http_methods(['GET', 'POST'])
def activate(request, token):
    invitation = _pending_invitation(token)
    if invitation is None or not invitation.is_active:
        return _render_private(request, 'accounts/staff_activation.html', {
            'invalid': True, 'message': GENERIC_ERROR,
        })
    if request.method == 'GET':
        return _render_private(request, 'accounts/staff_activation.html', {
            'invitation': invitation, 'email': invitation.email,
        })

    password1 = request.POST.get('new_password1') or ''
    password2 = request.POST.get('new_password2') or ''
    if password1 != password2:
        return _render_private(request, 'accounts/staff_activation.html', {
            'invitation': invitation, 'email': invitation.email,
            'error': 'Konfirmasi sandi tidak sama.',
        })
    try:
        user = accept_staff_invitation(token=token, password=password1)
    except ValidationError as exc:
        return _render_private(request, 'accounts/staff_activation.html', {
            'invitation': invitation, 'email': invitation.email,
            'error': ' '.join(exc.messages) or GENERIC_ERROR,
        })
    request.session['staff_activation_active'] = bool(user.is_active)
    return redirect('accounts:staff_activation_complete')


@require_GET
def activation_complete(request):
    active = request.session.pop('staff_activation_active', None)
    return _render_private(request, 'accounts/staff_activation_complete.html', {
        'active': active,
    })
```

Create both templates. `staff_activation.html` extends `base.html`, renders the
email as read-only text, two password inputs named
`new_password1`/`new_password2`, `{% csrf_token %}`, and either the
`error`/`message` alert or the form. `staff_activation_complete.html` renders
`Akun aktif. Silakan masuk.` and the login link only when `active is True`; for
`active is False`, render `Sandi berhasil dibuat. Akun menunggu aktivasi
administrator.` without a login promise; for a direct visit (`active is None`),
render only a generic completion heading. Neither template loads new external
assets, and the activation response's `Referrer-Policy: no-referrer` prevents
the path token from being sent as a referrer by assets inherited from
`base.html`.
